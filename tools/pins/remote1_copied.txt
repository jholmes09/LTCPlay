async function getJSON(p, ms){
  const ctl = new AbortController();
  const t = setTimeout(() => ctl.abort(), ms || 1500);
  try{
    const r = await fetch(p, {cache:"no-store", credentials:"same-origin", signal:ctl.signal});
    const j = await r.json().catch(() => ({}));
    if(!r.ok){ const e = new Error(j.error || r.statusText); e.status = r.status; throw e; }
    return j;
  } finally { clearTimeout(t); }
}
async function post(p, body){
  const r = await fetch(p, {method:"POST", credentials:"same-origin", cache:"no-store",
    headers:{"Content-Type":"application/json", "X-LTC-Window": WINDOW_ID}, body:JSON.stringify(body || {})});
  const j = await r.json().catch(() => ({}));
  if(!r.ok){ const e = new Error(refusedWords(j, r.statusText)); e.status = r.status; e.code = j.code || ""; throw e; }
  return j;
}

const SLOT_OPEN = new Set();
let SLOT_KEY = "";
function slotNotes(r){
  const out = [];
  if(r.reason) out.push(String(r.reason));
  if(r.planned && r.planned !== r.start) out.push("Planned for " + r.planned + ", moved to " + r.start + ".");
  if(r.origin === "edit") out.push("Added for tonight only.");
  if(r.origin === "operator") out.push("Started with Start now.");
  if(r.past_last_end) out.push("Runs past the night's last end time.");
  return out.join(" ");
}
function renderSchedule(slots){
  const key = JSON.stringify([slots.map(r => [r.show, r.start, r.status, slotNotes(r)]), [...SLOT_OPEN]]);
  if(key === SLOT_KEY) return;
  SLOT_KEY = key;
  $("tonight").innerHTML = slots.length ? slots.map(r => {
    const k = r.show + "@" + r.start, notes = slotNotes(r), open = SLOT_OPEN.has(k);
    return `<div class="slot${open ? " open" : ""}" data-k="${esc(k)}" role="button" tabindex="0" aria-expanded="${open}"><b>Show ${esc(r.show)}</b><span class="t">${esc(r.start)}</span><span class="s">${esc(r.status)}</span><span class="r">${esc(open ? (notes || "No notes for this show.") : notes)}</span></div>`;
  }).join("") : `<div class="dim">No shows tonight.</div>`;
}
function toggleSlot(k){
  if(SLOT_OPEN.has(k)) SLOT_OPEN.delete(k); else SLOT_OPEN.add(k);
  renderSchedule((ST && ST.schedule && ST.schedule.slots) || []);
}

async function armBeat(h, p){

  if(!h.all){

    if(!as.ok){ armStop(as.why); return; }

    const why = holdLapsed();

  }
  try{

    if(HOLD !== h || p.end) return;

function checkWords(f, j){

function pcZone(){ try{ return Intl.DateTimeFormat().resolvedOptions().timeZone || ""; }catch(e){ return ""; } }
function zoneLabel(z){
  for(const [, zs] of TZ_GROUPS) for(const [id, name] of zs) if(id === z) return name;
  return String(z || "").replace(/_/g, " ").replace(/\//g, " / ");
}
function allZones(){
  try{ return Intl.supportedValuesOf("timeZone"); }catch(e){ return []; }
}
function fillZones(want){
  const sel = $("se-tz");
  const known = new Set(TZ_GROUPS.flatMap(g => g[1].map(x => x[0])));
  const others = allZones().filter(z => !known.has(z));
  let h = "";
  if(want && !known.has(want) && !others.includes(want)) h += `<option value="${esc(want)}">${esc(zoneLabel(want))}</option>`;
  for(const [g, zs] of TZ_GROUPS) h += `<optgroup label="${esc(g)}">` + zs.map(([id, name]) => `<option value="${esc(id)}">${esc(name)}</option>`).join("") + `</optgroup>`;
  if(others.length) h += `<optgroup label="Every other zone">` + others.map(z => `<option value="${esc(z)}">${esc(zoneLabel(z))}</option>`).join("") + `</optgroup>`;
  sel.innerHTML = h;
  sel.value = want || "America/Denver";
  tzNow();
}

function dragMove(ev){
  if(!DRAG) return;
  if(DRAG.pointerId != null && ev.pointerId != null && ev.pointerId !== DRAG.pointerId) return;
  // A mouse that moves with no button down was let go somewhere this page
  // never heard: the drag is over, nothing follows a bare mouse.
  if(ev.pointerType === "mouse" && ev.buttons === 0){ dragEnd("the button was already up"); return; }
  const f = DRAG.field;
  const x = Math.max(0, Math.min(100, 100 * (ev.clientX - f.left) / f.width));
  const y = Math.max(0, Math.min(100, 100 * (ev.clientY - f.top) / f.height));
  DRAG.el.style.left = x + "%"; DRAG.el.style.top = y + "%";
  DRAG.pos = {x: Math.round(x * 10) / 10, y: Math.round(y * 10) / 10};
}
// The one way a drag ends, whatever ended it. `back` puts the machine

  if(ARRANGE) $("mo-msg").textContent = (back ? "Put back (" : "Dropped (") + why + "). Save layout keeps the new places; Cancel forgets them.";
}

// ---- Fireball (Jeff, 2026-10-28; design by Andy; fireball.py) --------
// One button per channel. Toggles (Fuel Charge, Air Charge, Purge) stay on
// until pressed again; Valve Open and Igniter are on only while a finger
// holds them. While anything is on the page beats every 100 ms; the engine
// and the flame link turn everything off 0.25 s after the last beat by
// themselves. Leaving the page, hiding it, closing it, a stale status,
// signing out and every refusal turn everything off here at once; losing
// focus lets go of Valve Open and Igniter. Nothing ever comes back on by
// itself: after a cut every channel has to be pressed again.
let FB = null, FB_OPEN = false, FB_TIMER = null, FB_BEAT = null, FB_RUN = null, FB_SEQ = 0, FB_RUNS = 0;
// Fuel or Air pressed but held by the engine for the charge record's mark
// (the beat's answer says held): the pad says WAITING, not ON.
let FB_HELD = [];
// The newest beat whose answer has been used: an older answer that arrives
// late (separate connections) must not turn the pad back.
let FB_ANS = 0;
function fbAnswer(run, seq, j){
  if(run !== FB_RUN || seq < FB_ANS) return;
  FB_ANS = seq;
  FB_HELD = Array.isArray(j && j.held) ? j.held : [];
  renderFireball();
}
const FB_ON = {fuel:false, air:false, valve:false, igniter:false, purge:false};
const FB_KEY = {};
function fireballView(on){
  if(on && !FB_OPEN){ FB_OPEN = true; fbPoll(); }
  if(!on && FB_OPEN){ FB_OPEN = false; clearTimeout(FB_TIMER); fbCut("the page was left"); }
}
async function fbPoll(){
  if(!FB_OPEN) return;
  try{
    if(!document.hidden){ FB = await getJSON("/api/remote/fireball", 1500); }
  }catch(e){ $("fb-msg").textContent = e.message; }
  renderFireball();
  FB_TIMER = setTimeout(fbPoll, 500);
}
function fbAny(){ return Object.keys(FB_ON).some(k => FB_ON[k]); }
function fbNow(){
  return fireballState(FB, FB_ON, isStale(Date.now(), LAST_OK, 1.0), !!(ST && ST.me && ST.me.view_only));
}
// Everything off here, now, and the engine told: this page's run ends and
// can never start again (an all off is never refused).
function fbCut(why){
  const had = fbAny(), run = FB_RUN;
  FB_HELD = [];
  for(const k in FB_ON) FB_ON[k] = false;
  FB_RUN = null; clearTimeout(FB_BEAT); FB_BEAT = null;
  if(run !== null) post("/api/remote/fireball-beat", actor(fireballBody(FB_ON, run, ++FB_SEQ, true))).catch(() => {});
  if(had) $("fb-msg").textContent = "Everything off: " + why;
  renderFireball();
}
// Let go of Valve Open and Igniter only (the window lost focus).
function fbRelease(why){
  if(!FB_ON.valve && !FB_ON.igniter) return;
  FB_ON.valve = false; FB_ON.igniter = false;
  $("fb-msg").textContent = "Valve Open and Igniter let go: " + why;
  fbSend(true);
}
// A change: sent at once. Nothing on: the run ends. Something on from
// everything off: a new run, named here.
function fbSend(change){
  clearTimeout(FB_BEAT); FB_BEAT = null;
  if(!fbAny()){ fbCut("everything was turned off"); return; }
  const st = fbNow();
  if(st.cut){ fbCut(st.cut); return; }
  if(FB_RUN === null) FB_RUN = WINDOW_ID + "-" + (++FB_RUNS) + "-" + Date.now().toString(36);
  renderFireball();
  fbBeat(change);
}
async function fbBeat(change){
  clearTimeout(FB_BEAT); FB_BEAT = null;
  if(!fbAny() || FB_RUN === null) return;
  const st = fbNow();
  if(st.cut){ fbCut(st.cut); return; }
  const run = FB_RUN, seq = ++FB_SEQ;
  FB_BEAT = setTimeout(() => fbBeat(false), 100);
  try{
    const j = await post("/api/remote/fireball-beat", actor(fireballBody(FB_ON, run, seq, change)));
    fbAnswer(run, seq, j);
  }
  catch(e){ if(run === FB_RUN) fbCut(e.message); }
}
function fbPress(k, on){
  if(FB_ON[k] === on) return;
  if(on){
    const st = fbNow();
    if(!st.ok){ $("fb-msg").textContent = st.block; return; }
  }
  FB_ON[k] = on;
  fbSend(true);
}
function renderFireball(){
  const v = FB, st = fbNow();
  if(st.cut && fbAny()){ fbCut(st.cut); return; }
  $("fb-armed").textContent = !v ? "Not known yet" : !v.available ? "Not available" :
    v.armed === "armed" ? "ARMED" + (v.group ? " (safety channel " + v.group.safety + ")" : "") :
    v.armed === "off" ? "Not armed" : "Armed: not known";
  $("fb-armed").className = "fb-armed" + (v && v.armed === "armed" ? " on" : "");
  const blk = st.block && st.block !== "Arm the Fireball first." ? st.block : "";
  $("fb-block").hidden = !blk; $("fb-block").textContent = blk;
  $("fb-note").textContent = st.text;
  $("fb-disarm").hidden = !(v && v.available);
  // The charge record does not know a tank (no record yet, a lost write):
  // an admin may confirm both are empty (fireball_charge.confirm_empty).
  const chk = ST && ST.fireball_charge, tks = (chk && chk.tanks) || {};
  // Also while automatic charging is off after a press that went out
  // unrecorded (auto_off), though every tank is known.
  $("fb-empty").hidden = !(v && v.available) || !(chk && chk.auto_off || Object.keys(tks).some(k => tks[k] && tks[k].state === "unknown"));
  const ctl = (v && v.controls) || [];
  const key = ctl.map(c => c.key + c.slot).join(",");
  const box = $("fb-pads");
  if(box.dataset.key !== key){
    box.dataset.key = key;
    box.innerHTML = ctl.map(c => `<button class="fb-pad" data-k="${esc(c.key)}" data-kind="${esc(c.kind)}"><b>${esc(c.label)}</b><span class="fb-st">OFF</span><small>${c.kind === "toggle" ? "Press: on, press again: off" : "Hold: on only while held"}, channel ${esc(c.slot)}</small><small class="fb-tk"></small></button>`).join("");
  }
  // The fuel and air charge (fireball_charge.py) and the timeline's skips
  // (fireball_cues.py), from every screen's status: display only.
  const ch = ST && ST.fireball_charge, tl = ST && ST.fireball_timeline;
  const said = [];
  if(ch) said.push(ch.fault ? "Charge record: " + ch.fault : (ch.text || ""));
  if(tl && tl.fault) said.push("Timeline: " + tl.fault);

  // One line per show whose latest play lost a shot to the Igniter rule.
  if(tl && tl.shots && tl.shots.length) said.push(tl.shots.join(" "));
  $("fb-charge").textContent = said.filter(Boolean).join(" ");
  const tanks = (ch && ch.tanks) || {};
  box.querySelectorAll(".fb-pad").forEach(b => {
    const on = !!FB_ON[b.dataset.k];
    b.classList.toggle("on", on);
    b.classList.toggle("na", !st.ok && !on);
    b.setAttribute("aria-pressed", on ? "true" : "false");
    b.setAttribute("aria-disabled", st.ok || on ? "false" : "true");

    if(tkEl) tkEl.textContent = !tk ? "" : tk.state === "charged" ? "Charged: a press is a top-off" : tk.state === "unknown" ? "Charge not known" : "";
  });
}

function wcLetGo(why){

  WC_HOLD = false; WC_RUN = null; clearTimeout(WC_BEAT); WC_BEAT = null;
  if(run !== null) post("/api/remote/cleaning-beat", actor(cleaningBody(false, run, ++WC_SEQ, true))).catch(() => {});

  const st = wcNow();
  if(!st.ok){ $("wc-msg").textContent = st.block; return; }

  WC_RUN = WINDOW_ID + "-c" + (++WC_RUNS) + "-" + Date.now().toString(36);

async function wcBeat(change){
  clearTimeout(WC_BEAT); WC_BEAT = null;
  if(!WC_HOLD || WC_RUN === null) return;
  const st = wcNow();
  if(st.cut){ wcLetGo(st.cut.replace(/[.\s]+$/, "")); return; }
  const run = WC_RUN, seq = ++WC_SEQ;
  WC_BEAT = setTimeout(() => wcBeat(false), 100);
  try{
    const j = await post("/api/remote/cleaning-beat", actor(cleaningBody(true, run, seq, change)));
    if(run !== WC_RUN || j.overtaken) return;
    WC = j;
    if(j.state === "phase1" || j.state === "phase2"){
      // The hold is complete: the routine runs on its own now.
      WC_HOLD = false; WC_RUN = null; clearTimeout(WC_BEAT); WC_BEAT = null;
      $("wc-msg").textContent = j.state === "phase1" ? "Phase 1 is running on its own. You can let go." :
        "The purge bursts are starting. You can let go; press Stop to end them.";
    } else if(!j.mine){ wcLetGo(j.last_note || j.last_cut || "the engine ended it"); return; }
    renderCleaning();
  }catch(e){ if(run === WC_RUN) wcLetGo(e.message.replace(/[.\s]+$/, "")); }
}

// ---- The timed purge (Jeff, 2026-10-29; morning.py) ---------------------
// The machines are selected exactly as for the Space bar's purge. The
// seconds: a preset or Other (a whole number, at most this show machine's
// longest). The button is HELD like a Hold to arm: the page beats every
// 100 ms and the engine starts the purge only after a whole hold of its
// own timing; a tap, letting go early, pointer cancel, losing focus,
// hiding or closing the page, a stale status, signing out and every
// refusal fire nothing. Once it runs you may let go: the engine ends it by
// itself, and its countdown is the engine's. Stop purge (here and on the
// banner on every screen) ends it now, as does Hold, Stop and any
// other purge press.
let TP_SECS = null, TP_HOLD = false, TP_RUN = null, TP_SEQ = 0, TP_RUNS = 0, TP_BEAT = null, TP_WIRED = false, TP_MSG = null;
// A line this page said (a refusal, the hold ended, started): shown for
// TP_SAY_MS, then the control's own state again.
const TP_SAY_MS = 6000;
function tpSay(text){ TP_MSG = {text:String(text || ""), at:Date.now()}; renderTimedPurge(); }
function tpSettings(){ return (MO && MO.timed_settings) || {presets:[2, 5, 10, 20, 30], default_s:5, max_s:30, min_s:1}; }
function tpNow(){
  return timedPurgeLook(MO, TP_SECS, TP_HOLD, isStale(Date.now(), LAST_OK, 1.0), !!(ST && ST.me && ST.me.view_only));
}
function tpLetGo(why){
  const had = TP_HOLD, run = TP_RUN;
  TP_HOLD = false; TP_RUN = null; clearTimeout(TP_BEAT); TP_BEAT = null;
  if(run !== null) post("/api/remote/purge-timed", moActor({hold:false, run:run, seq:++TP_SEQ, seconds:TP_SECS})).catch(() => {});
  if(had && why) TP_MSG = {text:"Hold ended: " + why + ". Nothing was fired.", at:Date.now()};
  renderTimedPurge();
}
function tpStart(){
  if(TP_HOLD) return;
  const st = tpNow();
  // A press on a timed purge that is running is a second purge press: it
  // ends it, as at the engine.
  if(st.running){ tpStopPress(); return; }
  if(arrangeBlocked("purge-fire", {})) return;
  if(!st.ok){ tpSay(st.block); return; }
  TP_HOLD = true;
  TP_RUN = WINDOW_ID + "-t" + (++TP_RUNS) + "-" + Date.now().toString(36);
    renderTimedPurge();
  tpBeat();
}
async function tpBeat(){
  clearTimeout(TP_BEAT); TP_BEAT = null;
  if(!TP_HOLD || TP_RUN === null) return;
  const st = tpNow();
  if(st.cut){ tpLetGo(st.cut.replace(/[.\s]+$/, "")); return; }
  const run = TP_RUN, seq = ++TP_SEQ;
  TP_BEAT = setTimeout(tpBeat, 100);
  try{
    const j = await post("/api/remote/purge-timed", moActor({hold:true, run:run, seq:seq, seconds:TP_SECS}));
    if(run !== TP_RUN || j.overtaken) return;
    if(j.started || j.running){
      // Started: the engine runs it to its end now; the finger may lift.
      TP_HOLD = false; TP_RUN = null; clearTimeout(TP_BEAT); TP_BEAT = null;
      TP_MSG = {text:(j.text || "Timed purge started.") + " You can let go.", at:Date.now()};
      try{ MO = await getJSON("/api/remote/morning", 1500); renderMorning(); }catch(e2){}
    }
    renderTimedPurge();
  }catch(e){ if(run === TP_RUN){ TP_HOLD = false; TP_RUN = null; clearTimeout(TP_BEAT); TP_BEAT = null; tpSay(e.message); } }
}
async function tpStopPress(){
  tpLetGo("");
  try{ const j = await post("/api/remote/purge-stop", moActor({})); TP_MSG = {text:j.text || "Purge stopped.", at:Date.now()}; }
  catch(e){ TP_MSG = {text:e.message, at:Date.now()}; }
  try{ MO = await getJSON("/api/remote/morning", 1500); renderMorning(); }catch(e2){}
}
function tpWire(){
  if(TP_WIRED) return;
  TP_WIRED = true;
  const hb = $("tp-hold");
  // Never the disabled attribute: a held button that turned disabled could
  // lose its lift. One that may not be pressed only says so.
  hb.addEventListener("pointerdown", e => { e.preventDefault(); try{ hb.setPointerCapture(e.pointerId); }catch(err){} tpStart(); });
  for(const ev of ["pointerup", "pointercancel", "lostpointercapture"])
    hb.addEventListener(ev, () => { if(TP_HOLD) tpLetGo("you let go before the hold was complete"); });
  hb.addEventListener("contextmenu", e => e.preventDefault());
  $("tp-stop").addEventListener("click", tpStopPress);
  $("tp-secs").addEventListener("change", () => {
    const ts = tpSettings();
    const v = timedClamp($("tp-secs").value, ts.min_s || 1, ts.max_s || 30);
    if(v === null){ $("tp-secs").value = TP_SECS === null ? "" : TP_SECS; return; }
    if(String(v) !== String($("tp-secs").value).trim()) TP_MSG = {text:"Kept to " + (ts.min_s || 1) + " to " + (ts.max_s || 30) + " s on this show machine.", at:Date.now()};
    TP_SECS = v; $("tp-secs").value = v;
    if(TP_HOLD) tpLetGo("the seconds changed");
    renderTimedPurge();
  });
}
window.addEventListener("blur", () => { if(TP_HOLD) tpLetGo("the page lost focus"); });
document.addEventListener("visibilitychange", () => { if(document.hidden && TP_HOLD) tpLetGo("the page was hidden"); });
window.addEventListener("pagehide", () => tpLetGo("the page closed"));
$("tp-banner-stop").addEventListener("click", tpStopPress);
function renderTimedPurge(){
  const box = $("tp"), mo = MO;
  box.hidden = !mo || !mo.timed_settings;
  if(box.hidden) return;
  tpWire();
  const ts = tpSettings();
  if(TP_SECS === null) TP_SECS = Math.min(Number(ts.default_s) || 5, Number(ts.max_s) || 30);
  const pk = JSON.stringify([ts.presets, ts.max_s]);
  const pb = $("tp-presets");
  if(pb.dataset.key !== pk){
    pb.dataset.key = pk;
    pb.innerHTML = (ts.presets || []).filter(n => n <= (ts.max_s || 30)).map(n => `<button class="ghost" data-s="${n}">${n} s</button>`).join("");
    pb.querySelectorAll("[data-s]").forEach(b => b.addEventListener("click", () => {
      if(TP_HOLD) tpLetGo("the seconds changed");
      TP_SECS = Number(b.dataset.s); renderTimedPurge();
    }));
    $("tp-secs").max = ts.max_s || 30; $("tp-secs").min = ts.min_s || 1;
  }
  pb.querySelectorAll("[data-s]").forEach(b => b.classList.toggle("on", Number(b.dataset.s) === TP_SECS));
  if(document.activeElement !== $("tp-secs")) $("tp-secs").value = TP_SECS;
  const st = tpNow();
  if(st.cut && TP_HOLD){ tpLetGo(st.cut.replace(/[.\s]+$/, "")); return; }
  const hb = $("tp-hold");
  hb.classList.toggle("on", TP_HOLD);
  hb.classList.toggle("run", st.running);
  hb.classList.toggle("na", !st.ok && !TP_HOLD && !st.running);
  hb.setAttribute("aria-pressed", TP_HOLD ? "true" : "false");
  hb.setAttribute("aria-disabled", !st.ok && !TP_HOLD ? "true" : "false");
  if(hb.textContent !== st.word) hb.textContent = st.word;
  const t = mo.timed || {};
  const said = TP_MSG && Date.now() - TP_MSG.at < TP_SAY_MS ? TP_MSG.text : "";
  const line = st.running ? "Purging " + (t.machines || []).join(", ") + " for " + t.seconds + " s: " + timedLeft(t.left_s) + " left. It stops by itself." :
    TP_HOLD ? "Keep holding, as for a Hold to arm." : said || st.block || t.last || "";
  if($("tp-st").textContent !== line){ $("tp-st").textContent = line; $("tp-st").title = line; }
}
